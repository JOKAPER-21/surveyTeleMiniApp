import os import tempfile from pathlib import Path import requests from flask import Flask, jsonify, request from werkzeug.utils import secure_filename app = Flask(__name__) # ========================================================= # CONFIGURATION # ========================================================= BOT_TOKEN = os.environ.get("BOT_TOKEN", "").strip() CHANNEL_ID = "-1004356863084" ALLOWED_EXTENSIONS = { "dwg", "dxf", "csv", "txt" } MAX_FILES = 10 # ========================================================= # HELPERS # ========================================================= def allowed_file(filename): if "." not in filename: return False extension = filename.rsplit(".", 1)[1].lower() return extension in ALLOWED_EXTENSIONS def telegram_api(method): return ( f"https://api.telegram.org/" f"bot{BOT_TOKEN}/" f"{method}" ) def build_caption( code, name, dgps_surveyor, command, date, time ): lines = [ "📍 DGPS SURVEY REPORT", "", f"🔖 CODE: {code}", f"📌 NAME: {name}", f"👷 DGPS SURVEYOR: {dgps_surveyor}", f"📅 DATE: {date}", f"🕒 TIME: {time}", "", "📝 COMMAND:", command if command else "-" ] return "\n".join(lines) # ========================================================= # HEALTH CHECK # ========================================================= @app.route("/", methods=["GET"]) def home(): return jsonify({ "status": "ok", "message": "VID Survey Telegram backend is running" }) # ========================================================= # UPLOAD TO TELEGRAM # ========================================================= @app.route("/upload", methods=["POST"]) def upload(): if not BOT_TOKEN: return jsonify({ "ok": False, "error": "BOT_TOKEN is not configured" }), 500 # ----------------------------------------------------- # GET FORM DATA # ----------------------------------------------------- code = request.form.get("code", "").strip() name = request.form.get("name", "").strip() dgps_surveyor = request.form.get( "dgps_surveyor", "" ).strip() command = request.form.get( "command", "" ).strip() date = request.form.get( "date", "" ).strip() time = request.form.get( "time", "" ).strip() # ----------------------------------------------------- # VALIDATION # ----------------------------------------------------- if not code: return jsonify({ "ok": False, "error": "Code is required" }), 400 if not name: return jsonify({ "ok": False, "error": "Name is required" }), 400 if not dgps_surveyor: return jsonify({ "ok": False, "error": "DGPS Surveyor is required" }), 400 uploaded_files = request.files.getlist("files") uploaded_files = [ file for file in uploaded_files if file and file.filename ] if not uploaded_files: return jsonify({ "ok": False, "error": "At least one file is required" }), 400 if len(uploaded_files) > MAX_FILES: return jsonify({ "ok": False, "error": ( f"Maximum {MAX_FILES} files allowed" ) }), 400 # ----------------------------------------------------- # CHECK FILE EXTENSIONS # ----------------------------------------------------- for file in uploaded_files: if not allowed_file(file.filename): return jsonify({ "ok": False, "error": ( f"File type not allowed: " f"{file.filename}" ) }), 400 # ----------------------------------------------------- # CREATE CAPTION # ----------------------------------------------------- caption = build_caption( code=code, name=name, dgps_surveyor=dgps_surveyor, command=command, date=date, time=time ) # ----------------------------------------------------- # TEMPORARY FILE FOLDER # ----------------------------------------------------- temp_dir = tempfile.mkdtemp( prefix="vidsurvey_" ) sent_files = [] try: # ------------------------------------------------- # SEND CAPTION FIRST # ------------------------------------------------- response = requests.post( telegram_api("sendMessage"), data={ "chat_id": CHANNEL_ID, "text": caption }, timeout=60 ) result = response.json() if not result.get("ok"): return jsonify({ "ok": False, "error": "Failed to send caption", "telegram": result }), 500 # ------------------------------------------------- # SAVE + SEND EACH FILE # ------------------------------------------------- for uploaded_file in uploaded_files: filename = secure_filename( uploaded_file.filename ) file_path = os.path.join( temp_dir, filename ) uploaded_file.save( file_path ) with open( file_path, "rb" ) as document_file: response = requests.post( telegram_api("sendDocument"), data={ "chat_id": CHANNEL_ID }, files={ "document": ( filename, document_file ) }, timeout=300 ) result = response.json() if not result.get("ok"): return jsonify({ "ok": False, "error": ( f"Failed to upload " f"{filename}" ), "telegram": result }), 500 sent_files.append( filename ) # ------------------------------------------------- # SUCCESS # ------------------------------------------------- return jsonify({ "ok": True, "message": ( "Survey successfully sent " "to Telegram channel" ), "files": sent_files }) except requests.RequestException as error: return jsonify({ "ok": False, "error": str(error) }), 500 finally: # ------------------------------------------------- # DELETE TEMPORARY FILES # ------------------------------------------------- try: for item in Path(temp_dir).iterdir(): if item.is_file(): item.unlink() Path(temp_dir).rmdir() except Exception as cleanup_error: print( "Cleanup error:", cleanup_error ) # ========================================================= # RUN SERVER # ========================================================= if __name__ == "__main__": print() print("=" * 50) print("VID SURVEY BACKEND STARTED") print("=" * 50) print("Server: http://0.0.0.0:5000") print("Upload endpoint: /upload") print("=" * 50) print() app.run( host="0.0.0.0", port=5000, debug=False )
